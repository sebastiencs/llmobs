### 2. `web/src/lib/roles.ts`

```ts
import type { Schema } from "@/api/types";

export type OrgRole = Schema<"OrgRole">;
export type RoleOverride = Schema<"ProjectRoleOverride">;

/** Highest first. Menus list roles in this order. */
export const ORG_ROLES: readonly OrgRole[] = ["owner", "admin", "member", "viewer"];

const RANK: Record<OrgRole, number> = { viewer: 0, member: 1, admin: 2, owner: 3 };

export function roleRank(role: OrgRole): number {
  return RANK[role];
}

/** False when `role` is undefined (an API key, or no membership). */
export function atLeast(role: OrgRole | undefined, min: OrgRole): boolean {
  return role !== undefined && RANK[role] >= RANK[min];
}

export function roleLabel(role: OrgRole): string {
  return role.charAt(0).toUpperCase() + role.slice(1);
}

/** Roles `actor` may give in an invite or a role change (D139): none above their own, `owner` only by an owner. */
export function grantableRoles(actor: OrgRole | undefined): OrgRole[] {
  if (!atLeast(actor, "admin") || actor === undefined) {
    return [];
  }
  return ORG_ROLES.filter(
    (role) => RANK[role] <= RANK[actor] && (role !== "owner" || actor === "owner"),
  );
}

/** Mirrors `may_set_role` in crates/server/src/routes/members.rs for the target's current role. */
export function canChangeRole(actor: OrgRole | undefined, target: OrgRole): boolean {
  if (!atLeast(actor, "admin")) {
    return false;
  }
  return target !== "owner" || actor === "owner";
}

/** Anyone may leave. Removing someone else mirrors `may_remove_other`. */
export function canRemoveMember(actor: OrgRole | undefined, target: OrgRole, self: boolean): boolean {
  if (self) {
    return true;
  }
  if (!atLeast(actor, "admin")) {
    return false;
  }
  return target !== "owner" || actor === "owner";
}

/** Overrides apply only to org members and viewers (D139). */
export function canHaveOverride(orgRole: OrgRole): boolean {
  return orgRole === "member" || orgRole === "viewer";
}

const OVERRIDE_LABELS: Record<RoleOverride | "inherit", string> = {
  inherit: "Inherit org role",
  admin: "Admin",
  member: "Member",
  viewer: "Viewer",
  none: "No access",
};

export function overrideLabel(value: RoleOverride | "inherit"): string {
  return OVERRIDE_LABELS[value];
}
```

### 3. `web/src/lib/redirect.ts`

```ts
const PUBLIC_PATHS: ReadonlySet<string> = new Set(["/login", "/signup", "/setup", "/reset", "/verify"]);

/** A relative path the server would also accept as `?redirect=` (`safe_redirect` in crates/server/src/auth/oidc/state_cookie.rs). */
export function safeRedirect(value: unknown): string | undefined {
  // The server's limit is 1024 bytes of UTF-8.
  if (typeof value !== "string" || value.length === 0 || new TextEncoder().encode(value).length > 1024) {
    return undefined;
  }
  if (!value.startsWith("/") || value.startsWith("//") || value.startsWith("/\\")) {
    return undefined;
  }
  for (let index = 0; index < value.length; index += 1) {
    const code = value.charCodeAt(index);
    if (code < 0x20 || code === 0x7f) {
      return undefined;
    }
  }
  return value;
}

/** Routes that render without a session (README § Routes). `/setup/workspace` is not one. */
export function isPublicPath(pathname: string): boolean {
  const path = pathname.length > 1 && pathname.endsWith("/") ? pathname.slice(0, -1) : pathname;
  return PUBLIC_PATHS.has(path) || path.startsWith("/invite/");
}

/** Full-page link that starts an OAuth/OIDC login (spec 08). */
export function authorizeHref(
  providerId: string,
  options: { redirect?: string; setupToken?: string } = {},
): string {
  const query = new URLSearchParams();
  if (options.redirect !== undefined) {
    query.set("redirect", options.redirect);
  }
  if (options.setupToken !== undefined) {
    query.set("setup_token", options.setupToken);
  }
  const base = `/api/auth/authorize/${encodeURIComponent(providerId)}`;
  const text = query.toString();
  return text.length > 0 ? `${base}?${text}` : base;
}
```

/** The first message for each field of a `400` problem; `{}` for anything else. */
export function fieldErrors(error: unknown): Record<string, string> {
  const problem = problemOf(error);
  const result: Record<string, string> = {};
  for (const item of problem?.errors ?? []) {
    if (result[item.field] === undefined) {
      result[item.field] = item.message;
    }
  }
  return result;
}

/** Text for a form-level error: the problem's detail, else its title, else `fallback`. */
export function problemMessage(error: unknown, fallback: string): string {
  const problem = problemOf(error);
  return problem?.detail ?? problem?.title ?? fallback;
}

import { Input } from "@/components/ui/input";

export function TextField({
  id,
  label,
  value,
  onChange,
  type = "text",
  error,
  description,
  autoComplete,
  required,
  disabled,
  readOnly,
  maxLength,
  placeholder,
}: {
  id: string;
  label: string;
  value: string;
  onChange: (value: string) => void;
  type?: "text" | "email" | "password" | "number";
  error?: string;
  description?: string;
  autoComplete?: string;
  required?: boolean;
  disabled?: boolean;
  readOnly?: boolean;
  maxLength?: number;
  placeholder?: string;
}) {

      <Label htmlFor={id}>{label}</Label>
      <Input
        id={id}
        name={id}
        type={type}
        value={value}
        onChange={(event) => {
          onChange(event.target.value);
        }}
        aria-invalid={error ? true : undefined}
        aria-describedby={describedBy.length > 0 ? describedBy : undefined}
        autoComplete={autoComplete}
        required={required}
        disabled={disabled}
        readOnly={readOnly}
        maxLength={maxLength}
        placeholder={placeholder}

        <p id={`${id}-description`} className="text-xs text-muted-foreground">
          {description}
        </p>
      ) : null}
      {error ? (
        <p id={`${id}-error`} className="text-xs text-destructive">
          {error}
        </p>
      ) : null}
    </div>
  );
}

  AlertDialogDescription,
  AlertDialogFooter,

} from "@/components/ui/alert-dialog";

export function ConfirmDialog({
  open,
  onOpenChange,
  title,
  description,
  confirmLabel,
  destructive,
  pending,
  error,
  onConfirm,
}: {
  open: boolean;
  onOpenChange: (open: boolean) => void;
  title: string;
  description: ReactNode;
  confirmLabel: string;
  destructive?: boolean;
  pending: boolean;
  error?: string;

    <AlertDialog

      onOpenChange={(next) => {
        // Esc must not close the dialog while the request runs.

        <AlertDialogHeader>
          <AlertDialogTitle>{title}</AlertDialogTitle>

            variant={destructive ? "destructive" : "default"}
            disabled={pending}

/** Wraps a disabled control so keyboard and screen-reader users still learn why (D-L-4). */
export function DisabledReason({
  reason,
  children,

      <span id={id} hidden>
        {reason}
      </span>
    </>
  );
}

import { HttpResponse } from "msw";

const TITLES: Record<number, string> = {
  400: "Bad Request",
  401: "Unauthorized",
  403: "Forbidden",
  404: "Not Found",
  409: "Conflict",
  429: "Too Many Requests",
  500: "Internal Server Error",
  503: "Service Unavailable",
};

/** A problem+json response like the server's (spec 09 § Conventions). */
export function problemResponse(
  status: number,
  code: string,
  options: { detail?: string; errors?: { field: string; message: string }[]; title?: string } = {},
): Response {
  return HttpResponse.json(
    {
      type: "about:blank",
      title: options.title ?? TITLES[status] ?? "Error",
      status,
      code,
      ...(options.detail !== undefined ? { detail: options.detail } : {}),
      ...(options.errors !== undefined ? { errors: options.errors } : {}),
    },
    { status, headers: { "Content-Type": "application/problem+json" } },
  );
}

### 4. `secret-dialog.tsx`

Behaviour (D-L-5):
- It is a controlled shadcn `Dialog`. Use `DialogContent` with `showCloseButton={false}`, so there is no corner X, and `className={showSnippets ? "sm:max-w-2xl" : undefined}` so the snippets fit.
- The header has `DialogTitle` = `title` and a `DialogDescription` that always ends with the sentence "Copy it now. You won't see it again.". When `description` is given, it comes first: `{description ? `${description} ` : ""}Copy it now. You won't see it again.`.
- The body is a `<div className="flex min-w-0 flex-col gap-3">`. Radix moves focus to the first focusable element when the dialog opens: the read-only input, which selects its text.
- Body:
  1. The label "Secret" (`<Label htmlFor="secret-value">`), then a row with a read-only `Input` (`id="secret-value"`, `value={secret}`, `className="font-mono text-xs"`, `onFocus={(event) => event.currentTarget.select()}`) and `<CopyButton value={secret} label="secret" onCopied={() => setCopied(true)} />`.
  2. When `showSnippets` is true and `secret.startsWith("sk-")`: the text "Send traces with this key:" (`text-sm`), then `<SnippetTabs origin={window.location.origin} apiKey={secret} />`.
  3. When `confirmingClose` is true: a bordered block (`role="alert"`, `rounded-md border border-destructive/40 p-3 text-sm`) with the text "You haven't copied the secret. Close without copying?" and two buttons, "Close anyway" (`variant="destructive"`, calls `finish()`) and "Go back" (`variant="outline"`, sets `confirmingClose` to false).
- Footer: one `Button` "Done" (`type="button"`) that calls `finish()`.
- State: `copied` (initially `false`) and `confirmingClose` (initially `false`). They must start fresh whenever `secret` changes, without an effect (README § Pitfalls): the exported `SecretDialog(props)` only returns `<SecretDialogBody key={props.secret} {...props} />`, and the private `SecretDialogBody` in the same file holds the state and everything below.
- Only the Copy button next to the secret counts as a copy. The snippet tabs' own copy buttons do not: a user who copied only a snippet is still asked "Close without copying?", because the snippet is not the secret on its own line.
- `finish()` sets both back to `false` and calls `onDone()`.
- Closing paths:
  - `DialogContent`'s `onEscapeKeyDown` and `onInteractOutside` handlers do only this: `if (!copied) { event.preventDefault(); setConfirmingClose(true); }`. They never call `finish()` themselves.
  - `Dialog`'s `onOpenChange={(next) => { if (!next) finish(); }}`. Radix calls it when Esc or an outside click is not prevented, so `finish()` runs exactly once.

```tsx
export function SecretDialog({
  open,
  title,
  secret,
  description,
  showSnippets,
  onDone,
}: {
  open: boolean;
  title: string;
  secret: string;
  description?: string;
  showSnippets?: boolean;
  onDone: () => void;
}) { … }
```

The dialog keeps the secret only in its props. Callers render it only while they have the secret, so `open` is always `true` there and nothing flashes empty while it closes:

```tsx
{create.data ? (
  <SecretDialog open title="…" secret={create.data.secret} onDone={() => create.reset()} />
) : null}
```

`create.reset()` clears the mutation's data, which unmounts the dialog (D-L-5).
